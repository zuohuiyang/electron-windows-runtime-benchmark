const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),assert=require('node:assert/strict');
const root=__dirname,read=p=>JSON.parse(fs.readFileSync(p,'utf8').replace(/^\uFEFF/,'')),cfg=read(path.join(root,'config.json'));
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
function protocol(){assert.equal(cfg.samples.length,80);assert.equal(new Set(cfg.samples.map(s=>s.id)).size,80);assert.equal(new Set(cfg.samples.map(s=>s.profile)).size,80);for(const drive of ['C','E']){for(const v of ['A','C'])assert.equal(cfg.samples.filter(s=>s.drive===drive&&s.variant===v).length,20);let afirst=0;for(let pair=1;pair<=20;pair++){const group=cfg.samples.filter(s=>s.drive===drive&&s.pair===pair);assert.equal(group.length,2);assert.deepEqual(group.map(s=>s.variant).sort(),['A','C']);afirst+=group[0].variant==='A';}assert.equal(afirst,10);}}
function verify(){protocol();const list=read(path.join(root,'inputs.json'));for(const f of list)assert(fs.statSync(f.path).size===f.size&&hash(f.path)===f.sha256,'Changed input: '+f.path);console.log('Verified '+list.length+' inputs and 80-sample balanced protocol');}
function validate(item,boots=new Set()){const base=path.join(root,'results',item.id),context=read(base+'-context.json');assert.equal(context.user,'autotest');assert.equal(context.sample.runtime,item.runtime);assert.equal(context.sample.profile,item.profile);assert.equal(context.sample.id,item.id);return require('./harness/validate.cjs').validate(read(base+'-sample.json'),context,read(base+'-result.json'),item,boots);}
const quantile=(values,p)=>{const a=[...values].sort((a,b)=>a-b),i=(a.length-1)*p,lo=Math.floor(i);return a[lo]+(a[Math.ceil(i)]-a[lo])*(i-lo);};
function report(){protocol();const boots=new Set(),rows=cfg.samples.map(s=>validate(s,boots)),stats=[];let seed=20260914;const random=()=>{seed=(1664525*seed+1013904223)>>>0;return seed/4294967296;};for(const drive of ['C','E']){const a=rows.filter(r=>r.drive===drive&&r.variant==='A'),c=rows.filter(r=>r.drive===drive&&r.variant==='C'),diff=a.map(r=>c.find(s=>s.pair===r.pair).firstFrameMs-r.firstFrameMs),boot=[];for(let b=0;b<10000;b++)boot.push(quantile(Array.from({length:20},()=>diff[Math.floor(random()*20)]),.5));const summarize=group=>({n:group.length,medianMs:quantile(group.map(r=>r.firstFrameMs),.5),p10Ms:quantile(group.map(r=>r.firstFrameMs),.1),p90Ms:quantile(group.map(r=>r.firstFrameMs),.9),minMs:Math.min(...group.map(r=>r.firstFrameMs)),maxMs:Math.max(...group.map(r=>r.firstFrameMs))});const A=summarize(a),C=summarize(c);stats.push({drive,A,C,medianReductionPercent:(1-C.medianMs/A.medianMs)*100,pairedMedianDeltaMs:quantile(diff,.5),pairedMedianDelta95PercentBootstrapMs:[quantile(boot,.025),quantile(boot,.975)],pairsFaster:diff.filter(d=>d<0).length});}
const laterFrameCallbacks=rows.filter(r=>r.presentedFrames>1).map(r=>({id:r.id,presentedFrames:r.presentedFrames}));
const warnings=cfg.samples.map(s=>({id:s.id,stderr:read(path.join(root,'results',s.id+'-sample.json')).stderr})).filter(s=>s.stderr);
fs.writeFileSync(path.join(root,'results','summary.json'),JSON.stringify({samples:80,independentBoots:boots.size,stats,laterFrameCallbacks,warnings,rows},null,2));const lines=['Electron overnight benchmark COMPLETE: 80 / 80','Completed: '+new Date().toISOString(),'20 launches per drive/variant; process creation -> first observed visible-video callback.','Runtime on named drive; OS, video, profiles and logs on C. Default Windows caching retained.','Independent reboot for each sample; 10 AC + 10 CA pairs per drive.',''];for(const s of stats)lines.push(`${s.drive}: upstream median ${s.A.medianMs.toFixed(3)} ms, split+preread median ${s.C.medianMs.toFixed(3)} ms; reduction ${s.medianReductionPercent.toFixed(2)}%`,`  upstream P10/P90 ${s.A.p10Ms.toFixed(3)}/${s.A.p90Ms.toFixed(3)} ms; split P10/P90 ${s.C.p10Ms.toFixed(3)}/${s.C.p90Ms.toFixed(3)} ms`,`  faster in ${s.pairsFaster}/20 pairs; paired C-A median ${s.pairedMedianDeltaMs.toFixed(3)} ms, bootstrap 95% [${s.pairedMedianDelta95PercentBootstrapMs.map(n=>n.toFixed(3)).join(', ')}] ms`);lines.push('',`Callbacks with frame count > 1 retained: ${laterFrameCallbacks.length}; samples with stderr: ${warnings.length}`,'No more automatic reboots. Switch to your usual account for analysis.','Results describe this machine/workload. Repeated reboot does not prove uncached storage.');fs.writeFileSync(path.join(root,'RESULTS.txt'),lines.join('\r\n'));console.log(lines.join('\n'));}
const [action,id]=process.argv.slice(2);if(action==='verify')verify();else if(action==='validate'){const item=cfg.samples.find(s=>s.id===id);assert(item);console.log(JSON.stringify(validate(item)));}else if(action==='report')report();else if(action==='protocol'){protocol();console.log('Protocol valid');}else throw Error('Unknown action');
