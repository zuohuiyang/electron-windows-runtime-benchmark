const fs=require('node:fs'),path=require('node:path'),crypto=require('node:crypto'),assert=require('node:assert/strict');
const root=__dirname,cold='C:/ElectronBench/host-callback-20x4-20260914';
const read=p=>JSON.parse(fs.readFileSync(p,'utf8').replace(/^\uFEFF/,''));
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex').toUpperCase();
const cfg=()=>read(path.join(root,'config.json'));
function walk(dir){return fs.readdirSync(dir,{withFileTypes:true}).flatMap(e=>{assert(!e.isSymbolicLink());const p=path.join(dir,e.name);return e.isDirectory()?walk(p):[p]});}
function protocol(){const c=cfg();for(const phase of ['warmup','measured'])for(const drive of ['C','E'])for(const variant of ['A','C'])assert.equal(c.actions.filter(x=>x.phase===phase&&x.drive===drive&&x.variant===variant).length,phase==='warmup'?5:20);assert.equal(new Set(c.actions.map(x=>x.profile)).size,100);assert.equal(new Set(c.actions.map(x=>x.id)).size,100);for(const drive of ['C','E']){let firstA=0;for(let pair=1;pair<=20;pair++){const a=c.actions.filter(x=>x.phase==='measured'&&x.drive===drive&&x.pair===pair);assert.deepEqual(a.map(x=>x.variant).sort(),['A','C']);firstA+=a[0].variant==='A';}assert.equal(firstA,10);}}
function prepare(){protocol();assert(!fs.existsSync(path.join(root,'initial-profile-hashes.json')),'Preparation already exists');const records=[];for(const a of cfg().actions){assert(!fs.existsSync(a.profile));for(const src of walk(a.seedProfile)){const dest=path.join(a.profile,path.relative(a.seedProfile,src));fs.mkdirSync(path.dirname(dest),{recursive:true});fs.copyFileSync(src,dest,fs.constants.COPYFILE_EXCL);assert.equal(hash(src),hash(dest));records.push({path:dest,size:fs.statSync(dest).size,sha256:hash(dest)});}}fs.writeFileSync(path.join(root,'initial-profile-hashes.json'),JSON.stringify(records,null,2));}
function pristine(){for(const f of read(path.join(root,'initial-profile-hashes.json')))assert.equal(hash(f.path),f.sha256,'Profile changed: '+f.path);}
function freeze(){const files=[...walk(path.join(root,'harness')),...['config.json','workflow.cjs','run-warm.ps1','run-at-logon.ps1','system-load.ps1','desktop.cs','progress.ps1','PROTOCOL.md','initial-profile-hashes.json'].map(f=>path.join(root,f))];const runtimes=read(path.join(cold,'inputs.json')).filter(f=>f.path.includes('\\runtimes\\'));const coldFiles=['run-at-logon-v2.ps1','controller-v2/system-load.ps1','harness/launch.cjs','harness/validate.cjs','pilot.cjs','config.json','inputs.json'].map(f=>path.join(cold,f));const list=[...runtimes,...[...files,...coldFiles].map(p=>({path:p,size:fs.statSync(p).size,sha256:hash(p)}))];fs.writeFileSync(path.join(root,'inputs.json'),JSON.stringify(list,null,2));}
function verify(){protocol();for(const f of read(path.join(root,'inputs.json')))assert(fs.statSync(f.path).size===f.size&&hash(f.path)===f.sha256,'Input changed: '+f.path);}
function validate(a){const base=path.join(root,'results',a.id),context=read(base+'-context.json'),raw=read(base+'-sample.json');assert.equal(context.user,'autotest');assert.equal(context.mode,'warm');for(const key of ['id','runtime','profile','phase'])assert.equal(context.sample[key],a[key]);assert.equal(raw.stderr,'');/* Same boot is intentional for warm samples: uniqueness is checked by sample id, not boot. */return require('./harness/validate.cjs').validate(raw,context,{status:'SUCCESS',inputsVerifiedAfter:true},a,new Set());}
const q=(values,p)=>{const a=[...values].sort((a,b)=>a-b),i=(a.length-1)*p,l=Math.floor(i);return a[l]+(a[Math.ceil(i)]-a[l])*(i-l);};

Warm uses same endpoint and independent seed clones,5 warmups per condition,same-session launches.',stats,rows};fs.writeFileSync(path.join(root,'SUMMARY.json'),JSON.stringify(summary,null,2));const fmt=n=>n.toFixed(3),change=x=>(x.ms>=0?'+':'')+fmt(x.ms)+' ms ('+(x.percent>=0?'+':'')+x.percent.toFixed(2)+'%)';const lines=['# Electron unified benchmark COMPLETE','', '160 formal launches: 20 per storage/temperature/variant. Both endpoints start at native process creation.','Cold 76 previous +4 resumed; warm80 new. Historical warm v4 excluded. See PROTOCOL.md for method differences.','', '| Storage | Condition | Endpoint | Baseline P50 | Split+preread P50 | Change | Baseline P90 | Split+preread P90 | Change |','|---|---|---|---:|---:|---|---:|---:|---|',...stats.map(s=>`| ${s.storage} | ${s.temperature} | ${s.endpoint==='appReadyMs'?'APP READY':'Visible video callback'} | ${fmt(s.A.p50)} | ${fmt(s.C.p50)} | ${change(s.p50Change)} | ${fmt(s.A.p90)} | ${fmt(s.C.p90)} | ${change(s.p90Change)} |`),'','No more automatic measurements or reboots. Switch to your usual account for analysis.','Cold means physical-host reboot-first launch with default Windows caching. Warm results are from a separate batch.'];fs.writeFileSync(path.join(root,'RESULTS.md'),lines.join('\r\n'));fs.writeFileSync(path.join(root,'RESULTS.txt'),lines.join('\r\n'));}
