$ErrorActionPreference='Stop';$root=$PSScriptRoot
Add-Type -AssemblyName System.Windows.Forms;Add-Type -AssemblyName System.Drawing
$form=New-Object Windows.Forms.Form;$form.Text='Electron benchmark - RUNNING';$form.ClientSize=New-Object Drawing.Size(680,230);$form.StartPosition='CenterScreen'
$label=New-Object Windows.Forms.Label;$label.Location=New-Object Drawing.Point(20,20);$label.Size=New-Object Drawing.Size(640,160);$label.Font=New-Object Drawing.Font('Segoe UI',12);$form.Controls.Add($label)
$bar=New-Object Windows.Forms.ProgressBar;$bar.Location=New-Object Drawing.Point(20,190);$bar.Size=New-Object Drawing.Size(640,20);$bar.Maximum=100;$form.Controls.Add($bar)
$timer=New-Object Windows.Forms.Timer;$timer.Interval=5000
$refresh={
 try {
  $s=Get-Content "$root\warm-state.json" -Raw|ConvertFrom-Json;$c=Get-Content "$root\control.json" -Raw|ConvertFrom-Json
  $pre=@($s.completed|Where-Object {$_ -like 'prep-*'}).Count;$formal=@($s.completed|Where-Object {$_ -like 'warm-*'}).Count
  $bar.Value=[Math]::Min(100,$pre+$formal)
  if($s.status -eq 'COMPLETE' -and (Test-Path "$root\SUMMARY.json")){$form.Text='Electron benchmark - COMPLETE';$label.Text="ALL TESTS COMPLETE. You may switch accounts now.`r`n`r`nCold: 80/80 (previously completed)`r`nWarm: 80/80; warmups: 20/20`r`nResults: $root\RESULTS.txt";$timer.Stop()}
  elseif($s.status -in @('FAILED','PAUSED_ENVIRONMENT','STOPPED') -or $c.status -eq 'FAILED'){$form.Text='Electron benchmark - STOPPED';$label.Text="TEST STOPPED. This is not a complete result.`r`n`r`nWarmups: $pre/20; formal warm samples: $formal/80`r`n$($s.error)`r`nYou may switch accounts for diagnosis.";$timer.Stop()}
  else{$label.Text="TEST RUNNING - please do not switch accounts or lock the screen.`r`n`r`nCold: 80/80 complete. Warm testing is still in progress.`r`nWarmups: $pre/20; formal warm samples: $formal/80`r`nState: $($s.status)`r`nClosing this progress window does not stop the test."}
 }catch{$label.Text='Waiting for benchmark state. Please remain in this account.'}
}
$timer.Add_Tick($refresh);$form.Add_Shown({& $refresh;$timer.Start()});$form.Add_FormClosed({$timer.Stop();$timer.Dispose()});[void]$form.ShowDialog()
